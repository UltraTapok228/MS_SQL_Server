### Триггеры для таблицы
```sql
USE College2;
GO

-- 1 Триггер вида INSERT (срабатывает после добавления нового студента)
CREATE OR ALTER TRIGGER trg_AfterInsert_Student
ON Students
AFTER INSERT
AS
BEGIN
    DECLARE @NewName NVARCHAR(100);
    SELECT @NewName = FullName FROM inserted;
    
    PRINT 'Успех! В систему добавлен новый студент: ' + @NewName;
END;
GO

-- 2 Триггер вида UPDATE
CREATE OR ALTER TRIGGER trg_AfterUpdate_Student
ON Students
AFTER UPDATE
AS
BEGIN
    IF UPDATE(GroupId)
    BEGIN
        PRINT 'Внимание: Зафиксирован перевод студента в другую группу!';
    END
    ELSE
    BEGIN
        PRINT 'Данные студента были обновлены.';
    END
END;
GO

-- 3 Триггер вида DELETE
CREATE OR ALTER TRIGGER trg_PreventDelete_Student
ON Students
INSTEAD OF DELETE
AS
BEGIN
    PRINT 'ОШИБКА: Удаление записей из таблицы Students запрещено политикой безопасности!';
END;
GO
```

## "Провокация" триггеров
```sql
USE College2;
GO

-- 1 Провоцируем триггер INSERT (добавляем тестового студента)
INSERT INTO Students (FullName, GroupId, EnrolledYear) 
VALUES ('Тестов Тест Тестович', 1, 2026);

-- 2 Провоцируем триггер UPDATE (переводим его в группу 2)
UPDATE Students 
SET GroupId = 2 
WHERE FullName = 'Тестов Тест Тестович';

-- 3 Провоцируем триггер DELETE (пытаемся его удалить - триггер должен это запретить)
DELETE FROM Students 
WHERE FullName = 'Тестов Тест Тестович';
```

### Триггеры для БД
```sql
USE College2;
GO

-- 1 Триггер БД на создание новых таблиц (Логирование)
CREATE OR ALTER TRIGGER trg_DB_CreateTable
ON DATABASE
FOR CREATE_TABLE
AS
BEGIN
    -- вывод уведомления каждый раз, когда кто-то пишет CREATE TABLE
    PRINT 'Событие БД: В базе данных College2 была создана новая таблица!';
END;
GO

-- 2. Триггер БД на удаление таблиц (Защита)
CREATE OR ALTER TRIGGER trg_DB_PreventDropTable
ON DATABASE
FOR DROP_TABLE
AS
BEGIN
    PRINT 'КРИТИЧЕСКАЯ ОШИБКА: Удаление таблиц в этой базе запрещено!';
    -- Откатываем транзакцию, не давая удалить таблицу
    ROLLBACK; 
END;
GO
```
